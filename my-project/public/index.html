async function startPayment(amount, name, phone, email = "", method = "upi") { const res = await fetch("/api/create-payment", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ amount, customer_name: name, customer_phone: phone, customer_email: email, method, }), }); const data = await res.json(); if (data.success && data.checkout_url) { window.location.href = data.checkout_url; } else { alert(data.error || "Payment start nahi ho paya"); } }
